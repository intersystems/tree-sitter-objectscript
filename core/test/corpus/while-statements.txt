 WHILE x<10 {
   WRITE !," Looping",x
   SET x=x+1
 }

 SET mylist=$LISTBUILD("a","b","c","d","e")
 SET ptr=0,sampcnt=1,sampmax=4
 WHILE 1=$LISTNEXT(mylist,ptr,value),sampcnt<sampmax {
   WRITE value," is item ",sampcnt,!
   SET sampcnt=sampcnt+1
 }

 WHILE x < 10
   {
      WRITE !,"Looping",x
      QUIT:x=5
      SET x=x+1
   }

 SET x=1,y=1
 WHILE x<6 {
   WRITE "outer loop ",!
   WHILE y<100 {
     WRITE "inner loop "
     WRITE " y=",y,!
     QUIT:y=7
     SET y=y+2
   }
   WRITE "back to outer loop x=",x,!!
   SET x=x+1
 }
  WRITE "Done"
