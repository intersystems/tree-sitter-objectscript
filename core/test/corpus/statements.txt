			Set:mode'="" lock = lock _ "#""" _ mode _ """"

		Set $ZE="<SYNTAX>"
		Goto Trap
	}

	If timeout'="" {
	   Set lock = lock _ ":"_timeout
	} Else {
	   Set $TEST = 1
	}

	// If anything is invalid, this will raise a <SYNTAX> error which we'll convert to
	// a type error.
	LOCK @lock

	Quit $S($TEST:$system.Python.True(),1:$system.Python.False())
