USE Valid Keywords
===================

  use device:/NEW
  use device:/READ
  use device:/WRI
  use device:/APP
  use device:/APPEND
  use device:/STREAM
  use device:/VARIABLE
  use device:/FIX
  use device:/FIXED
  use device:/UNDEFINED
  use device:(/TRANSLATE=0:/IOTABLE="test")
  use device:(/TRA=0:/IOT="test")
  use device:/XYT="test"
  use device:/XYTABLE="test"
  use device:/NOXY
  use device:/NOXY=2
  use device:/OBUFSIZE=2
  use device:/GZIP=2
  use device:/GZIP
  use device:/COMPRESS="hi"
  use device:/PAR="hi"
  use device:/PARAMS="hi"
  use device:/REC="hi"
  use device:/WRITE
  use device:(/CRE)
  use device:/CREATE
  use device:/TERMINATOR="test"
  use device:(/TER="test")
  use device:/delete
  use device:/del=2
