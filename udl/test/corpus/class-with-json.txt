    ClassMethod %ElementSchema(irisType As %String, ByRef visited) As %DynamicObject [Internal]
    {
        if iristype = "" Return {"type": "string"} 
        set u = $ZCONVERT(irisType, "U")

        set clsType = $$$getClassType(irisType) 
        if clsType = $$$cCLASSCLASSTYPEDATATYPE {
            return ..%JsonType(irisType)
        } 
        if clsType = $$$cCLASSCLASSTYPESTREAM {
            if irisType [ "Binary" Return {"type": "string", "contentEncoding": "base64"}
            Return {"type": "string"}
        } 
        if ##class(%Dictionary.CompiledClass).%ExistsId(irisType) {
            Return ..%ClassSchema(irisType, .visited)
        }
        Return {"type": "string"} 
    }
}
